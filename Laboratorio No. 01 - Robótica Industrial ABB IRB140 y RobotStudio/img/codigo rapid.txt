    PROC FarmAura()
    ! =========================================
    ! FASE 1: El "Mewing" (Postura de superioridad)
    ! =========================================
    ! El robot sube y se echa hacia adelante, sacando "el pecho"
    MoveJ Offs(Home_robot, 0, -130, 80), v100, fine, dfernandSim\WObj:=wobj0;
    WaitTime 1.5; ! Pausa dramática para mirar a los demás a los ojos
    
    ! =========================================
    ! FASE 2: El "Dab" (Movimiento rápido y gracioso)
    ! =========================================
    ! Giro brusco hacia la derecha simulando el dab
    MoveJ Offs(Home_robot, 130, -80, 0), v500, fine, dfernandSim\WObj:=wobj0;
    WaitTime 0.8; 
    
    ! Se cruza rápido hacia la izquierda
    MoveJ Offs(Home_robot, -130, -80, 0), v500, fine, dfernandSim\WObj:=wobj0;
    WaitTime 0.4;

    ! =========================================
    ! FASE 3: El "6 - 7" (El meme viral)
    ! =========================================
    ! Balanceos rápidos de izquierda a derecha
    MoveJ Offs(Home_robot, -100, -30, -50), v500, fine, dfernandSim\WObj:=wobj0;
    WaitTime 0.15;
    MoveJ Offs(Home_robot, 100, -30, -50), v500, fine, dfernandSim\WObj:=wobj0;
    WaitTime 0.15;
    MoveJ Offs(Home_robot, -100, -30, -50), v500, fine, dfernandSim\WObj:=wobj0;
    WaitTime 0.15;
    MoveJ Offs(Home_robot, 100, -30, -50), v500, fine, dfernandSim\WObj:=wobj0;
    WaitTime 0.4;

    ! =========================================
    ! FASE 4: "Farming Aura" (El remate final)
    ! =========================================
    ! El robot se eleva al centro y se queda quieto, imponiendo respeto
    MoveJ Offs(Home_robot, 0, -50, 130), v200, fine, dfernandSim\WObj:=wobj0;
    WaitTime 2.0; ! Pausa larga de "Aura Farming"
    
    ! =========================================
    ! FASE 5: Regreso a casa
    ! =========================================
    Path_home;
ENDPROC

    PROC DrawSegment(num x1, num y1, num x2, num y2)
    ! Lift tool (Negative Z goes UP)
    MoveL Offs(Target_80, x1, y1, -15), v100, fine, dfernandSim\WObj:=wobj1_tablero;
    ! Lower to surface
    MoveL Offs(Target_80, x1, y1, 0), v50, fine, dfernandSim\WObj:=wobj1_tablero;
    ! Draw line
    MoveL Offs(Target_80, x2, y2, 0), v50, fine, dfernandSim\WObj:=wobj1_tablero;
    ! Lift tool
    MoveL Offs(Target_80, x2, y2, -15), v100, fine, dfernandSim\WObj:=wobj1_tablero;
ENDPROC

PROC DrawCircle(num cx, num cy, num r)
    VAR num i;
    VAR num ang;
    VAR num x;
    VAR num y;

    ! Move to start of circle
    MoveL Offs(Target_80, cx + r, cy, -15), v100, fine, dfernandSim\WObj:=wobj1_tablero;
    MoveL Offs(Target_80, cx + r, cy, 0), v50, fine, dfernandSim\WObj:=wobj1_tablero;

    ! Draw circle
    FOR i FROM 1 TO 36 DO
        ang := i * 10;
        x := cx + r * Cos(ang);
        y := cy + r * Sin(ang);
        MoveL Offs(Target_80, x, y, 0), v50, fine, dfernandSim\WObj:=wobj1_tablero;
    ENDFOR

    ! Lift tool
    MoveL Offs(Target_80, cx + r, cy, -15), v100, fine, dfernandSim\WObj:=wobj1_tablero;
ENDPROC
